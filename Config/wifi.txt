# ===== 机型默认值：局域网 / 无线 =====
# 优先级：workflow 显式传入 WRT_* > 机型专属段 > 全局段 > Settings.sh 内置兜底
# 键名前缀 = 设备键（WRT_CONFIG 去掉最后一个 -发行版 后缀，去非字母数字后转大写）
# 例：TR3000-immortalwrt -> TR3000
#
# 说明：
# - 本方案不使用 Router Tool 的 wireless defaults（它强制 country=US、
#   encryption=psk2+ccmp、2.4G 20MHz、5G 后缀为 5.2G/5.8G）。
#   所有无线参数由 Settings.sh 生成的 10-wifi-defaults 全量写入。
#   因此 WIFI_COUNTRY 可自由设为 CN 或 US，不会被覆盖。
# - WIFI_*_MODE 目前仅支持 AX（WiFi6→HE）和 BE（WiFi7→EHT）；
#   加 AC（WiFi5→VHT）/ N（WiFi4→HT）机型需同步扩展 Settings.sh 的 case 分支。
# - LAN_PASSWORD 仅用于构建信息展示，不会写入实际 root 密码。

# ================= 全局兜底 =================
LAN_IP=192.168.10.1
LAN_PASSWORD=无
HOST_NAME=OWRT

WIFI_SSID=OWRT
WIFI_PASSWORD=12345678
WIFI_ENCRYPTION=sae-mixed
WIFI_COUNTRY=US
WIFI_2G_MODE=AX
WIFI_5G_MODE=AX
WIFI_2G_WIDTH=40
WIFI_5G_WIDTH=160

TR3000_WIFI_5G_MODE=AX
